# ---- LA REGLE DE POSE ----
#
# LE RELIEF. Une hauteur par case : `~` le vide, `0..9` les etages. Le
# hors-carte est du VIDE, jamais de la plaine -- un moteur qui lirait
# l'absence comme de la plaine etendrait le sol a l'infini et ne poserait
# aucune falaise de bord du monde.
void -1 out_of_map_is_void
plain level_like_any_other
#
# LE MASSIF. Pour chaque etage L a partir de ZERO, une composante connexe
# des cases de niveau AU MOINS L -- pas exactement L. Les massifs sont donc
# EMBOITES : une case de niveau 3 appartient aux massifs 3, 2, 1 et 0.
# Groupes a l'etage exact, un porteur deviendrait un ANNEAU troue la ou le
# massif du dessus se pose, et l'on verrait le sol a travers son milieu.
massif connected_component level_at_least from_zero
#
# L'EMPREINTE. La surface du massif, PLUS la falaise qui pend sous chaque
# bord sud. La silhouette se lit sur l'EMPREINTE, jamais sur la surface
# seule : une case de falaise a des voisines de falaise, et la lire dans la
# surface la dirait isolee.
footprint surface_plus_south_skirt
silhouette read_on_footprint eight_bits blob_canonical
#
# LA CHUTE. Un etage vaut TROIS rangees. La face cherche le premier palier j
# tel que le sol, 3j rangees plus bas, soit exactement j etages sous la
# levre ; elle n'est donc PAS forfaitaire. Si elle n'en trouve aucun, elle
# tombe dans le vide et NE POSE PAS SON PIED -- on ne touche pas le fond
# d'un gouffre. Et la sonde regarde SOUS LE PIED, jamais les rangees
# traversees : une terrasse posee sur le trajet n'arrete pas la chute.
fall three_rows_per_storey probe_under_foot no_foot_into_void
#
# LES BANDES. La surface, puis le chapiteau au premier pas, le corps
# ensuite, le pied a l'atterrissage. Une tuile EST le couple (silhouette,
# bande) -- plus le seul bit de contexte qui reste.
bands surface crown body foot
#
# LE SEUL BIT DE CONTEXTE : EST-CE QUE JE RECOUVRE UN AUTRE MASSIF ?
# Si oui, mon contour est AU RAS partout -- son arete touche la bordure de
# tuile au rendez-vous, pour que n'importe quelle voisine puisse la
# rejoindre. Si non, mes aretes sont LIBRES. La PLAINE ne compte pas comme
# chose recouverte : sans cette clause tout massif recouvrirait quelque
# chose, donc tout serait au ras, et la liberte disparaitrait.
flush covers_another_massif plain_does_not_count
#
# L'ORDRE DE POSE. Par RANGEE d'abord, par ETAGE ensuite ; la derniere pose
# gagne. Un chevauchement n'est QUE de l'occlusion par ordre de rendu : le
# massif du dessus masque la falaise du dessous, il ne la modifie pas. Le
# voisinage d'une tuile ne depend donc jamais de ce qu'on pose dessus.
draw_order by_row_then_level last_wins
#
# ⚠ ET IL N'Y A NI POSE ARRIERE NI SOUS-PIED. Puisque le massif du dessus
# est au ras partout, celui du dessous n'emerge jamais EN PLEIN MILIEU d'une
# tuile : il emerge toujours a une bordure. Il n'existe donc aucune tuile
# composite, et le massif du dessous est dessine COMPLET.
no_backing no_underfoot massif_drawn_whole
#
# L'EPAISSEUR. Aucune course de cases du meme etat -- pleine ou vide -- ne
# fait moins de TROIS, dans chaque silhouette d'etage. Ce n'est pas une
# regle en soi : c'est la consequence d'un pinceau qui pose un tampon 3x3.
thin_forbidden run_shorter_than_three all_storeys both_states
#
sheet_rule fixed_size one_drawing_per_tile no_variants
